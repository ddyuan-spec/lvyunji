① 代码（整段替换节点内代码）

你面板里现在那段是旧版（params['userId'] != null ? …），请用下面这段重新覆盖一次，否则输入名叫 userid 的会取不到值。

async function main({ params }) {
  const userId = String(params['userId'] || params['user_id'] || params['userid'] || '').trim();
  // 商品识别 LLM 在「生成多样性=平衡」时会给商品名加句号/空格（如「惠民补贴套餐包4。」），
  // 后端按名查是精确/模糊匹配、带标点就查不到 → 这里兜底把首尾标点空白全清掉。
  const productName = String(params['productName'] || params['product_name'] || '')
    .replace(/^[\s﻿.。,，、；;：:！!？?·\-_~～]+/, '')
    .replace(/[\s﻿.。,，、；;：:！!？?·\-_~～]+$/, '')
    .trim();
  const env = String(params['env'] || 'test');

  if (!userId) return { success: false, error: '缺少 user_id', product: null, card_json: '' };
  if (!productName) return { success: false, error: '缺少 productName（商品名）', product: null, card_json: '' };

  const HOST = env === 'prod'
    ? 'https://yxm-api.juhuanvip.com'
    : 'https://yxm-test-api.juhuanvip.com';
  const IMG_BASE = 'https://yxm-test-api.juhuanvip.com'; // 图片域名前缀，待后端给真实值再改
  const ACCESS_TOKEN = 'db776bc648144a819a7850df4fd7673d';

  const form = new URLSearchParams();
  form.append('type', '2');            // 帮我买 = 按商品名查单个
  form.append('userId', userId);
  form.append('productName', productName);

  const resp = await fetch(HOST + '/api/apiWxStore/v2//coze/free/assistProduct', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded',
      'accessToken': ACCESS_TOKEN,
      'wxName': 'lym',
      'apiFrom': 'H5MALL'
    },
    body: form.toString()
  });
  const json = await resp.json();
  const data = json.data;
  const raw = Array.isArray(data) ? (data[0] || null) : data;
  if (json.code !== 1 || !raw) return { success: false, error: (json.msg) || '未查到该商品', product: null, card_json: '' };

  const img = (raw.productMainImg || '').startsWith('http')
    ? raw.productMainImg
    : IMG_BASE + (raw.productMainImg || '');

  const product = {
    productId: raw.productMainId,
    name: raw.productName,
    image: img,
    priceText: raw.discountPrice != null ? '¥' + raw.discountPrice : '',
    originalText: raw.originalPrice != null ? '¥' + raw.originalPrice : '',
    saveText: raw.priceDifference ? '省¥' + raw.priceDifference : '',
    scoreText: raw.score != null ? 'AI评分 ' + raw.score : '',
    desc: raw.desc || ''
  };
  const card_json = JSON.stringify(product);
  return { success: true, error: '', product, card_json };
}
