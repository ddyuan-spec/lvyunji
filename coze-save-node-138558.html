① 代码（整段替换节点内代码）

async function main({ params }) {
  // 兼容 userId / user_id / userid 三种输入名
  const userId = String(params['userId'] != null ? params['userId']
                : (params['user_id'] != null ? params['user_id']
                : (params['userid'] || ''))).trim();
  const env = String(params['env'] || 'test');
  // 帮我省 = save → 0；帮我省选 = choose → 1（都返回数组）
  const intent = String(params['custom_variables'] || params['intent'] || 'save').trim().toLowerCase();
  const type = (intent === 'choose' || intent === '1') ? '1' : '0';

  if (!userId) return { success: false, error: '缺少 user_id', products: [], card_json: '', summary: '', count: 0 };

  const HOST = env === 'prod' ? 'https://yxm-api.juhuanvip.com' : 'https://yxm-test-api.juhuanvip.com';
  const IMG_BASE = 'https://yxm-test-api.juhuanvip.com'; // 图片域名前缀，待后端给真实值再改
  const ACCESS_TOKEN = 'db776bc648144a819a7850df4fd7673d';

  const form = new URLSearchParams();
  form.append('type', type);   // 0 = 帮我省（按差额排序的商品数组）
  form.append('userId', userId);

  const resp = await fetch(HOST + '/api/apiWxStore/v2//coze/free/assistProduct', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded',
      'accessToken': ACCESS_TOKEN,
      'wxName': 'lym',
      'apiFrom': 'H5MALL'
    },
    body: form.toString()
  });
  const json = await resp.json();
  const list = Array.isArray(json.data) ? json.data : (json.data ? [json.data] : []);
  if (json.code !== 1 || list.length === 0)
    return { success: false, error: (json.msg) || '未查到省钱商品', products: [], card_json: '', summary: '', count: 0 };

  const products = list.map(raw => {
    const img = (raw.productMainImg || '').startsWith('http')
      ? raw.productMainImg : IMG_BASE + (raw.productMainImg || '');
    return {
      productId: raw.productMainId,
      name: raw.productName,
      image: img,
      priceText: raw.discountPrice != null ? '¥' + raw.discountPrice : '',
      originalText: raw.originalPrice != null ? '¥' + raw.originalPrice : '',
      saveText: raw.priceDifference ? '省¥' + raw.priceDifference : '',
      scoreText: raw.score != null ? 'AI评分 ' + raw.score : '',
      desc: raw.desc || ''
    };
  });
  const card_json = JSON.stringify(products);                 // 多卡片：JSON 数组
  const totalSave = list.reduce((s, r) => s + (Number(r.priceDifference) || 0), 0);
  const summary = '为你找到 ' + products.length + ' 件可省钱的商品，预计共省¥' + totalSave;
  return { success: true, error: '', products, card_json, summary, count: products.length };
}
