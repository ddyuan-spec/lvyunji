<script>
// 会员等级配置（演示用）
const memberLevel = '店主';
const REWARD_THRESHOLD = 5000;
const REWARD_RATE = 0.02;

// 打开审批弹窗
function openModal() {
    document.getElementById('auditModal').classList.add('show');
    // 根据会员等级判断是否显示奖励
    if (memberLevel === '店主') {
        document.getElementById('rewardInfo').classList.remove('hidden');
    }
}

// 关闭审批弹窗
function closeModal() {
    document.getElementById('auditModal').classList.remove('show');
}

// 计算奖励金额
function calculateReward() {
    const amount = parseFloat(document.getElementById('depositAmount').value) || 0;
    let reward = 0;
    
    if (memberLevel === '店主' && amount >= REWARD_THRESHOLD) {
        reward = amount * REWARD_RATE;
    }
    
    const total = amount + reward;
    
    document.getElementById('showDeposit').textContent = '¥' + amount.toFixed(2);
    document.getElementById('rewardAmount').textContent = '¥' + reward.toFixed(2);
    document.getElementById('totalAmount').textContent = '¥' + total.toFixed(2);
}

// 打开二次确认弹窗
function openConfirmModal() {
    const amount = parseFloat(document.getElementById('depositAmount').value) || 0;
    document.getElementById('confirmInput').value = amount.toFixed(2);
    updateConfirmDetail();
    document.getElementById('confirmModal').classList.add('show');
}

// 关闭二次确认弹窗
function closeConfirmModal() {
    document.getElementById('confirmModal').classList.remove('show');
}

// 更新确认弹窗详情
function updateConfirmDetail() {
    const amount = parseFloat(document.getElementById('confirmInput').value) || 0;
    let reward = 0;
    
    if (memberLevel === '店主' && amount >= REWARD_THRESHOLD) {
        reward = amount * REWARD_RATE;
    }
    
    const total = amount + reward;
    
    document.getElementById('confirmDeposit').textContent = '¥' + amount.toFixed(2);
    document.getElementById('confirmReward').textContent = '¥' + reward.toFixed(2);
    document.getElementById('confirmTotal').textContent = '¥' + total.toFixed(2);
}

// 提交确认
function submitConfirm() {
    const amount = parseFloat(document.getElementById('confirmInput').value) || 0;
    alert('提交成功！实际充值金额：¥' + amount.toFixed(2));
    closeConfirmModal();
    closeModal();
}

        e.target.classList.remove('show');
    }
}
