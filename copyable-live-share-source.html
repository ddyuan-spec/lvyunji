<script>
var currentLink = '';

function openShareModal(liveId, link) {
    currentLink = link;
    document.getElementById('shareModal').classList.add('show');
}
function closeShareModal() {
    document.getElementById('shareModal').classList.remove('show');
}

function sharePoster() {
    closeShareModal();
    showToast('生成海报中...');
    // 实际业务中这里跳转到海报生成页
}

function openLinkModal() {
    closeShareModal();
    document.getElementById('shareLinkText').textContent = currentLink;
    document.getElementById('linkModal').classList.add('show');
    // 重置复制按钮状态
    var btn = document.getElementById('copyBtn');
    btn.classList.remove('copied');
    btn.innerHTML = '<svg width="16" height="16" viewBox="0 0 24 24" fill="currentColor"><path d="M16 1H4c-1.1 0-2 .9-2 2v14h2V3h12V1zm3 4H8c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h11c1.1 0 2-.9 2-2V7c0-1.1-.9-2-2-2zm0 16H8V7h11v14z"/></svg>复制链接';
}
function closeLinkModal() {
    document.getElementById('linkModal').classList.remove('show');
}

function copyLink() {
    var link = document.getElementById('shareLinkText').textContent;
    if (navigator.clipboard) {
        navigator.clipboard.writeText(link).then(function() {
            onCopySuccess();
        }).catch(function() {
            fallbackCopy(link);
        });
    } else {
        fallbackCopy(link);
    }
}

function fallbackCopy(text) {
    var textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed';
    textarea.style.opacity = '0';
    document.body.appendChild(textarea);
    textarea.select();
    try {
        document.execCommand('copy');
        onCopySuccess();
    } catch (e) {
        showToast('复制失败，请长按链接手动复制');
    }
    document.body.removeChild(textarea);
}

function onCopySuccess() {
    var btn = document.getElementById('copyBtn');
    btn.classList.add('copied');
    btn.textContent = '已复制';
    showToast('链接已复制，快去分享给好友吧');
}

function showToast(msg) {
    var t = document.getElementById('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(function() { t.classList.remove('show'); }, 2000);
}

// 点击遮罩关闭弹窗
document.getElementById('shareModal').addEventListener('click', function(e) {
    if (e.target === this) closeShareModal();
});
document.getElementById('linkModal').addEventListener('click', function(e) {
    if (e.target === this) closeLinkModal();
});
</script>
